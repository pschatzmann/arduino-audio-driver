28// Make it easier to represent bits from the datasheet.
29// Bits D[l] - D[r] = 0b[v] l & r are between 7 and 0 inclusive.
30// For example: "D7-D5 set to 010" is `makeBits(7, 5, 0b010)` or
31// `makeBits(7, 5, 2)`.
32// Plain functions (not macros) are used here, unlike ESP-IDF's own
33// single-arg BIT(x) macro (from esp_bit_defs.h), so this header never has
34// to undef/restore anyone else's macro. They live at namespace scope
35// (rather than as class members) so they are fully defined before any
36// static constexpr class member that uses them in its initializer.
37static constexpr uint8_t makeBits(uint8_t l, uint8_t r, uint8_t v) {
38 return (uint8_t)(((v) << (r)) & ((1 << ((l) + 1)) - 1));
39}
40static constexpr uint8_t makeBit(uint8_t l, uint8_t v) {
41 return (uint8_t)(((v) & 0x01) << (l));
42}

386 bool configureClocks(uint32_t sample_rate, uint8_t word_size, uint8_t channels, uint32_t mclk, bool is_master) {
387 AD_LOGI("TLV320AIC3104 configureClocks() rate:%d mclk:%d is_master:%s", sample_rate, mclk, is_master?"true":"false");
388
389 // Are we using MCLK or BCLK to derive our local clock?
390 bool use_bclk = (mclk == 0);
391 uint32_t clock = use_bclk ? (sample_rate * word_size * channels) : mclk;
392 AD_LOGI("MCLK: %d Using %s at %dHz", mclk, use_bclk?"BCLK":"MCLK", clock);
393
394 // Find the appropriate PLL and other register parameters for our clockrate.
395 size_t rate_count = 0;
396 const RateDivs* rate_table = rateDivTable(rate_count);
397 const RateDivs* rate_entry = nullptr;
398 for (size_t i = 0; i < rate_count; i++) {
399 if (rate_table[i].rate == sample_rate) {
400 rate_entry = &rate_table[i];
401 break;
402 }
403 }
404 if (rate_entry == nullptr) {
405 AD_LOGE("TLV320AIC3104::configureClocks() No rate_entry found");
406 return false;
407 }
408 AD_LOGI("Rate Table entry found: fSref:%d NCODEC:%d", rate_entry->fsref, rate_entry->ncodec_code);
409
410 size_t pll_count = 0;
411 const PllDivs* pll_table = pllDivTable(pll_count);
412 const PllDivs* pll_entry = nullptr;
413 for (size_t i = 0; i < pll_count; i++) {
414 if (pll_table[i].clk == clock && pll_table[i].fsref == rate_entry->fsref) {
415 pll_entry = &pll_table[i];
416 break;
417 }
418 }
419 if (pll_entry == nullptr) {
420 AD_LOGE("TLV320AIC3104::configureClocks() No pll_entry found. clock:%d fSref:%d", clock, rate_entry->fsref);
421 return false;
422 }
423 AD_LOGI("PLL Div entry found: D:%d J:%d P:%d R:%d", pll_entry->d, pll_entry->j, pll_entry->p, pll_entry->r);
424
425 bool rc = true;
426
427 // D7 is PLL enable. D6-3 is Q. D2-0 is P
428 uint8_t r3 = makeBit(7, 0b1) | makeBits(6, 3, (PLL_Q_CODE(2))) | makeBits(2, 0, PLL_P_CODE(pll_entry->p));
429 uint8_t r8 = makeBits(7, 6, is_master?0b11:0b00); // BCLK and LRCLK set to input (slave) or output (master)
430 // D0 is CODEC_CLKIN: 0 is PLL, 1 is CLKDIV
431 uint8_t r101 = makeBit(0, 0b0); // PLLDIV_OUT
432 uint8_t r102 = makeBits(7, 6, (use_bclk)?0b10:0b00) // CLKDIV_IN src. 0=MCLK, 2=BCLK.
433 | makeBits(5, 4, (use_bclk)?0b10:0b00) // PLLCLK_IN src. 0=MCLK, 2=BCLK.
434 | makeBits(3, 0, 0b0010); // reserved bits, required to be this value.
435 rc &= writePagedReg(PLL_PROGB_ADDR, makeBits(7, 2, pll_entry->j));
436 rc &= writePagedReg(PLL_PROGC_ADDR, makeBits(7, 0, pll_entry->d >> 6));
437 rc &= writePagedReg(PLL_PROGD_ADDR, makeBits(7, 2, pll_entry->d));
438 rc &= writePagedReg(OVRF_STATUS_PLLR_ADDR, makeBits(3, 0, PLL_R_CODE(pll_entry->r)));
439 rc &= writePagedReg(ASD_INTF_CTRLA_ADDR, r8);
440 rc &= writePagedReg(CLK_ADDR, r101);
441 rc &= writePagedReg(CLK_GEN_ADDR, r102);
442 rc &= writePagedReg(PLL_PROGA_ADDR, r3);
443 rc &= writePagedReg(ASD_INTF_CTRLA_ADDR, makeBit(5, 1)); // Hi-z on idle
444
445 // fS(ref) selection + shared NCODEC divider + dual-rate mode
446 uint8_t datapath = (rate_entry->fsref == 44100) ? FSREF_44100 : FSREF_48000;
447 datapath |= LDAC2LCH | RDAC2RCH;
448 if (rate_entry->dual_rate) datapath |= DUAL_RATE_MODE;
449 rc &= writePagedReg(CODEC_DATAPATH_ADDR, datapath);
450 rc &= writePagedReg(SAMPLE_RATE_SEL_ADDR,
451 makeBits(7, 4, rate_entry->ncodec_code) | // ADC Sample Rate Select
452 makeBits(3, 0, rate_entry->ncodec_code) // DAC Sample Rate Select
453 );
454 return rc;
455
456 }

534 bool rc = true;
535
536 // Default to powered up, but muted
537 uint8_t left1_reg = makeBits(6, 3, 0b1111) | makeBit(2, 1); // 19
538 uint8_t right1_reg = makeBits(6, 3, 0b1111) | makeBit(2, 1); // 22
539 uint8_t left2_reg = 0xFF; // 17

544 left1_reg =

546 makeBits(6, 3, 0) | // Input pad: -0dB
547 makeBit(2, 1) | // ADC powered up
548 makeBits(1, 0, 0); // ADC PGA soft stepping, 1/sample
549 right1_reg = left1_reg;
550 pga_gain_reg =
551 makeBit(7, 0) | // Not muted
552 makeBits(6, 0, 0b0110000); // 24dB PGA Gain about right for a microphone
553 break;
554 case ADC_INPUT_LINE1: // Single ended inputs on Line1/Mic1, Line level.
555 left1_reg =
556 makeBit(7, 0) | // Single ended input
557 makeBits(6, 3, 0) | // Input pad: -0dB
558 makeBit(2, 1) | // ADC powered up
559 makeBits(1, 0, 0); // ADC PGA soft stepping, 1/sample
560 right1_reg = left1_reg;
561 pga_gain_reg =

569 right2_reg =
570 makeBits(7, 4, 0b1111) | // Mic2L/Line2L -X-> Right ADC
571 makeBits(3, 0, 0b0000); // Mic2R/Line2R ---> Right ADC
572 pga_gain_reg =
573 makeBit(7, 0) | // Not muted
574 makeBits(6, 0, 0b0000000); // 0dB PGA Gain for line level input
575 break;
576 case ADC_INPUT_LINE3: // ADC_INPUT_LINE3: Single ended inputs on Line2/Mic2, Mic gain and bias.
577 left2_reg =
578 makeBits(7, 4, 0b0000) | // Mic2L/Line2L ---> Left ADC
579 makeBits(3, 0, 0b1111); // Mic2R/Line2R -X-> Left ADC
580 right2_reg =
581 makeBits(7, 4, 0b1111) | // Mic2L/Line2L -X-> Right ADC
582 makeBits(3, 0, 0b0000); // Mic2R/Line2R ---> Right ADC
583 pga_gain_reg =

585 makeBits(6, 0, 0b0110000); // 24dB PGA Gain about right for a microphone
586 break;
587 case ADC_INPUT_ALL: // Singled ended line inputs on Line1, Single ended mono Mic input on Mic2, Line2 off.
588 default:
589 left1_reg =
590 makeBit(7, 0) | // Single ended input
591 makeBits(6, 3, 0b1000) | // Input pad: -12dB to equalize Line Input with Mic gain, a little anyway.
592 makeBit(2, 1) | // ADC powered up
593 makeBits(1, 0, 0); // ADC PGA soft stepping, 1/sample
594 right1_reg = left1_reg;
595 left2_reg =
596 makeBits(7, 4, 0b0000) | // Mic2L/Line2L ---> Left ADC
597 makeBits(3, 0, 0b0000); // Mic2R/Line2R ---> Left ADC
598 right2_reg =
599 makeBits(7, 4, 0b1111) | // Mic2L/Line2L -X-> Right ADC
600 makeBits(3, 0, 0b1111); // Mic2R/Line2R -X-> Right ADC
601 pga_gain_reg =
602 makeBit(7, 0) | // Not muted
603 makeBits(6, 0, 0b0110000); // 24dB PGA Gain about right for a microphone

719 bool readPagedReg(RegAddr reg, uint8_t& value) {
720 if (!selectPage(reg.page)) return false;
721 // https://github.com/palmerr23/tlv320AIC3104/blob/2d53fc2c95a07494dad87241c47c2086245e0c8b/src/tlv320aic3104_comms.h#L91

724 // * ZephyrDriverCommon.h readReg()
725 // * API_I2C.h i2c_bus_read_bytes()

730 bool updatePagedReg(RegAddr reg, uint8_t mask, uint8_t value) {
731 uint8_t old = 0;
732 if (!readPagedReg(reg, old)) return false;
733 uint8_t updated = (old & ~mask) | (value & mask);
734 if (updated == old) return true;
735 return writePagedReg(reg, updated);
736 }

747 bool selectPage(uint8_t page) {
748 if (current_page == page) return true;
749 uint8_t reg = PAGE_CONTROL_ADDR;
750 if (!writeReg(reg, page)) return false;
751 // pg47, 10.5.2. Recommended to read the page register back after writing.
752 uint8_t new_page = 0xFF;
753 // Ignoring errors. I don't want to succeed in write, but not update current_page because
754 // of a read error. new_page will EITHER be the new page (the read) actually succeeded,
755 // or it'll still be 0xFF which will trigger an attempt to write again next time.
756 readReg(reg, new_page);
757 current_page = new_page;
758 return true;
759 }
